Decision Trees, CART, Boundary Analysis, Entropy & Random Forests

A read-only rendering of my decision-tree experiments, including explanations, code, and the saved plots and outputs.

This note brings together my experiments on decision trees: starting from CART and Gini impurity, looking at the resulting decision boundaries, then replacing Gini with entropy, and finally extending the idea toward random forests.

def GiniImpurity(values):
    G = 1
    s = sum(values)
    if s == 0:
        return 1
    for val in values:
        G -= (val/s)**2
    return G

In [3]:
def GiniSplit(ginis, n, N): # CART Cost Function
    g = 0
    for ginis_, n_ in zip(ginis, n):
        g += ginis_*n_/N
    return g

def split(X, Y):
    min_ginisplit = None
    thresh_group = {}
    best_ginil = None
    best_ginir = None
    best_n_left = None
    best_n_right = None
    best_values_l = None
    best_values_r = None
    best_class_l = None
    best_class_r = None
    for x_range in range(int(min(X)), int(max(X) * 100) + 1):
        values_l = {i:0 for i in list(set(Y))}
        values_r = {i:0 for i in list(set(Y))}
        #print(values_l, values_r)
        for x, y in zip(X, Y):
            if x <= x_range/100:
                values_l[y] += 1
            else:
                values_r[y] += 1
        ginil = GiniImpurity(list(values_l.values()))
        ginir = GiniImpurity(list(values_r.values()))
        n_left = sum(values_l.values())
        n_right = sum(values_r.values())
        ginisplit = GiniSplit([ginil, ginir],[n_left, n_right], len(X))
        if min_ginisplit is None or ginisplit <= min_ginisplit:
            min_ginisplit = ginisplit
            thresh = x_range/100
            if min_ginisplit in thresh_group:
                thresh_group[min_ginisplit].append(thresh)
            else:
                thresh_group[min_ginisplit] = [thresh]
                best_ginil = ginil
                best_ginir = ginir
                best_n_left = n_left
                best_n_right = n_right
                best_values_l = list(values_l.values())
                best_values_r = list(values_r.values())
                best_class_l = int(max(values_l, key=values_l.get))
                best_class_r = int(max(values_r, key=values_r.get))

    threshold = sum(thresh_group[min_ginisplit])/ len(thresh_group[min_ginisplit])
    return threshold, min_ginisplit, {"Left": {"gini": best_ginil, "samples": best_n_left, "values": best_values_l, "class": best_class_l, "confidence": max(best_values_l)/sum(best_values_l)},
                       "Right": {"gini": best_ginir, "samples": best_n_right, "values": best_values_r, "class": best_class_r,
                                 "confidence": max(best_values_r)/sum(best_values_r)}}

def DT(X_data, y_data, k, gini=None, depth=2):
    if gini == 0:
        return
    if not depth: return
    if not k:
        return
    mini_gini = None
    selected_k = None
    splits = None
    tk = None
    for feature in k:
        x_data = X_data[:, feature]
        tk_, gini_, splits_ = split(x_data, y_data)
        if mini_gini is None or mini_gini > gini_:
            mini_gini = gini_
            selected_k = feature
            tk = tk_
            splits = splits_
    left_split, right_split = splits["Left"], splits["Right"]
    print("Threshold : ", tk, "Gini :", mini_gini, "Feature :", selected_k)
    print("Left Split : ", left_split)
    print("Right SPlit : ", right_split)
    print()
    X_left_split_data = X_data[X_data[:, selected_k] <= tk]
    X_right_split_data = X_data[X_data[:, selected_k] > tk]
    y_left_split_data = y_data[X_data[:, selected_k] <= tk]
    y_right_split_data = y_data[X_data[:, selected_k] > tk]
    DT(X_left_split_data, y_left_split_data, k, left_split["gini"], depth-1)
    DT(X_right_split_data, y_right_split_data, k, right_split["gini"], depth-1)

def decision_boundary(tree, X, Y, X_test, y_test):
    t = tree.tree_
    x_min, x_max = X[:, 0].min(), X[:, 0].max()
    y_min, y_max = X[:, 1].min(), X[:, 1].max()
    plt.figure(figsize=(8, 6))
    def traverse(node, xlim, ylim):
        feature = t.feature[node]
        threshold = t.threshold[node]
        if feature == -2:
            predicted_class = np.argmax(t.value[node][0])
            color = "yellow" if predicted_class == 0 else "orange"
            plt.fill([xlim[0], xlim[1], xlim[1], xlim[0]], [ylim[0], ylim[0], ylim[1], ylim[1]], color=color, alpha=0.2)
            return
        if feature == 0:
            traverse(t.children_left[node], (xlim[0], threshold), ylim)
            traverse(t.children_right[node], (threshold, xlim[1]), ylim)
        elif feature == 1:
            traverse(t.children_left[node], xlim, (ylim[0], threshold))
            traverse(t.children_right[node], xlim, (threshold, ylim[1]))
    traverse(0, (x_min, x_max), (y_min, y_max))
    plt.scatter(X[Y == 0, 0], X[Y == 0, 1], color="blue", edgecolor="black", label="Class 0")
    plt.scatter(X[Y == 1, 0], X[Y == 1, 1], color="orange", edgecolor="black", label="Class 1")
    plt.scatter(X_test[y_test == 0, 0], X_test[y_test == 0, 1], color="green", edgecolor="white", label="Class test 0")
    plt.scatter(X_test[y_test == 1, 0], X_test[y_test == 1, 1], color="yellow", edgecolor="white", label="Class test 1")
    plt.xlabel("X1")
    plt.ylabel("X2")
    plt.title("Decision Boundary")
    plt.legend()
    plt.grid()
    plt.show()

def Impurity(values):
    G = 0
    s = sum(values)
    if s == 0:
        return 1
    for val in values:
        if val and s:
            G -= (val/s) * math.log2(val/s)
    return G

def Split(entropies, n, N): # CART Cost Function
    g = 0
    for entropy, n_ in zip(entropies, n):
        g += entropy*n_/N
    return g
